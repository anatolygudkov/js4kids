Конец игры

В настоящей игре можно проиграть, иначе неинтересно. Наша змейка будет погибать, если врежется в стену или укусит себя за хвост. А на поле появится надпись «Игра окончена».

В этом уроке ты:
  • заменишь порталы на настоящие стены;
  • напишешь функцию isOnSnake, которая проверяет, занята ли клеточка змейкой;
  • узнаешь, как остановить таймер (clearInterval) и функцию (return);
  • напишешь на холсте текст.

Остановить таймер

Змейку двигает таймер setInterval. Чтобы игра остановилась, таймер надо выключить. Для этого его нужно запомнить в переменной. Измени последнюю строчку программы:

setInterval(step, 200);
let timer = setInterval(step, 200);

setInterval возвращает номер таймера, и мы кладём его в timer. Выключить таймер можно командой clearInterval(timer).

Функция «Игра окончена»

Добавь новую функцию после draw:

function endGame() {
  clearInterval(timer);
  ctx.fillStyle = "white";
  ctx.font = "32px sans-serif";
  ctx.textAlign = "center";
  ctx.fillText("Игра окончена", 200, 200);
}
  • clearInterval(timer) — выключить таймер: step больше не вызывается, змейка замирает;
  • ctx.font — размер и вид шрифта: 32 пикселя, шрифт без засечек (sans-serif);
  • ctx.textAlign = "center" — выравнивать текст по центру;
  • ctx.fillText(текст, x, y) — написать текст в точке (x, y). Точка (200, 200) — середина поля.
💡 Проверь в консоли Набери в консоли endGame() и нажми Enter: змейка остановится и появится надпись. Так можно проверять функции по одной, ещё до того, как они подключены к игре.

Стены

Порталы были весёлыми, но в классической Змейке стены опасны. Убери из step четыре проверки-портала и вставь одну проверку стен:

function step() {
  let head = { x: snake[0].x + dx, y: snake[0].y + dy };

  if (head.x >= cells) {
    head.x = 0;
  }
  if (head.x < 0) {
    head.x = cells - 1;
  }
  if (head.y >= cells) {
    head.y = 0;
  }
  if (head.y < 0) {
    head.y = cells - 1;
  }
  if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
    endGame();
    return;
  }

  snake.unshift(head);

Через || («или») мы собрали четыре вопроса в один: «голова левее поля, или правее, или выше, или ниже?» Если хоть что-то из этого правда, игра окончена.

Зачем return?

return немедленно завершает функцию: строчки ниже уже не выполнятся. Без него змейка, уже «врезавшись», всё равно добавила бы голову за пределами поля и перерисовала бы картинку поверх надписи «Игра окончена».

Укусить себя за хвост

Нужно проверить, не попала ли новая голова на клеточку, где уже есть кусочек змейки. Для этого надо пройти по всем кусочкам циклом и сравнить координаты. Напишем функцию-вопрос, которая отвечает true или false:

function isOnSnake(x, y) {
  for (let i = 0; i < snake.length; i++) {
    if (snake[i].x === x && snake[i].y === y) {
      return true;
    }
  }
  return false;
}

Как это работает:

  • цикл по очереди смотрит на каждый кусочек змейки;
  • если нашёлся кусочек на клеточке (x, y), функция сразу отвечает true («да, тут змейка») и заканчивается;
  • если цикл проверил всех и никого не нашёл, программа доходит до последней строчки и отвечает false.

Имя функции похоже на вопрос: is on snake? — «на змейке?». Так принято называть функции, которые отвечают «да» или «нет».

Теперь используем её в step, сразу после проверки стен:

  if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
    endGame();
    return;
  }
  if (isOnSnake(head.x, head.y)) {
    endGame();
    return;
  }

Сохрани, обнови и проверь оба способа проиграть: врежься в стену, а потом отрасти змейку подлиннее и укуси себя. Чтобы сыграть снова, пока что нажимай F5. В следующем уроке сделаем перезапуск по пробелу.

📄 Весь файл snake.html после этого урока 
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Змейка</title>
</head>
<body>
  <h1>Змейка</h1>
  <p>Счёт: <span id="score">0</span></p>
  <canvas id="game" width="400" height="400"></canvas>
  <script>
    let canvas = document.getElementById("game");
    let ctx = canvas.getContext("2d");
    let size = 20;
    let cells = 20;

    let snake = [
      { x: 10, y: 10 },
      { x: 9, y: 10 },
      { x: 8, y: 10 }
    ];
    let apple = { x: 15, y: 10 };
    let dx = 1;
    let dy = 0;
    let score = 0;

    function clearField() {
      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 400, 400);
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * size, y * size, size - 1, size - 1);
    }

    function randomCell() {
      return Math.floor(Math.random() * cells);
    }

    function isOnSnake(x, y) {
      for (let i = 0; i < snake.length; i++) {
        if (snake[i].x === x && snake[i].y === y) {
          return true;
        }
      }
      return false;
    }

    function draw() {
      clearField();
      drawCell(apple.x, apple.y, "red");
      for (let i = 0; i < snake.length; i++) {
        drawCell(snake[i].x, snake[i].y, "lime");
      }
      drawCell(snake[0].x, snake[0].y, "yellow");
    }

    function endGame() {
      clearInterval(timer);
      ctx.fillStyle = "white";
      ctx.font = "32px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText("Игра окончена", 200, 200);
    }

    function step() {
      let head = { x: snake[0].x + dx, y: snake[0].y + dy };

      if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
        endGame();
        return;
      }
      if (isOnSnake(head.x, head.y)) {
        endGame();
        return;
      }

      snake.unshift(head);

      if (head.x === apple.x && head.y === apple.y) {
        score = score + 1;
        document.getElementById("score").textContent = score;
        apple.x = randomCell();
        apple.y = randomCell();
      } else {
        snake.pop();
      }

      draw();
    }

    function onKey(event) {
      if (event.key === "ArrowUp" && dy !== 1) {
        dx = 0;
        dy = -1;
      }
      if (event.key === "ArrowDown" && dy !== -1) {
        dx = 0;
        dy = 1;
      }
      if (event.key === "ArrowLeft" && dx !== 1) {
        dx = -1;
        dy = 0;
      }
      if (event.key === "ArrowRight" && dx !== -1) {
        dx = 1;
        dy = 0;
      }
    }

    document.addEventListener("keydown", onKey);
    draw();
    let timer = setInterval(step, 200);
  </script>
</body>
</html>
