Новая игра

Проигрыш — и надо нажимать F5… Неудобно! Сделаем, как в настоящих играх: нажал пробел — и новая игра. Для этого немного перестроим программу. А ещё запретим яблоку появляться внутри змейки.

В этом уроке ты:
  • соберёшь всё, что нужно для начала игры, в функцию startGame;
  • добавишь переменную-флажок gameOver;
  • сделаешь перезапуск по пробелу;
  • познакомишься с циклом while.

Что значит «начать заново»

Когда начинается новая игра, надо:

  1. вернуть змейку из трёх кусочков в середину поля;
  2. направить её вправо;
  3. обнулить счёт;
  4. положить яблоко в новое место;
  5. нарисовать поле и запустить таймер.

Почти всё это сейчас делается в самом начале программы, когда мы создаём переменные. Перенесём это в функцию startGame («начать игру»), чтобы её можно было вызывать сколько угодно раз.

💡 Это называется рефакторинг Когда программисты перестраивают код, чтобы он стал удобнее, но игра при этом работает так же, это называется рефакторинг. Это нормальная часть работы. Не торопись и после каждого шага проверяй, что игра запускается.

Шаг 1. Пустые коробки

Переменные оставим наверху, но значения будем класть в них внутри startGame. Если написать let без =, получится пустая коробка. Добавим ещё две новые: gameOver и timer.

let canvas = document.getElementById("game");
let ctx = canvas.getContext("2d");
let size = 20;
let cells = 20;

let snake = [
  { x: 10, y: 10 },
  { x: 9, y: 10 },
  { x: 8, y: 10 }
];
let apple = { x: 15, y: 10 };
let dx = 1;
let dy = 0;
let score = 0;
let snake;
let apple = { x: 0, y: 0 };
let dx;
let dy;
let score;
let gameOver;
let timer;

gameOver («игра окончена») — это флажок: в нём будет лежать true или false. По нему программа узнаёт, идёт игра или уже закончилась.

Шаг 2. Функция startGame

Добавь её в самый конец, после функции onKey:

function startGame() {
  snake = [
    { x: 10, y: 10 },
    { x: 9, y: 10 },
    { x: 8, y: 10 }
  ];
  dx = 1;
  dy = 0;
  score = 0;
  gameOver = false;
  showScore();
  placeApple();
  draw();
  timer = setInterval(step, 200);
}

А последние строчки программы станут такими:

document.addEventListener("keydown", onKey);
draw();
let timer = setInterval(step, 200);
startGame();

Заметь: внутри startGame мы пишем snake = … и timer = … без let, потому что коробки уже созданы наверху. Если поставить let, получится новая коробка, которая живёт только внутри функции, а остальная программа её не увидит.

Шаг 3. Новые помощники

В startGame мы позвали две функции, которых ещё нет: showScore и placeApple. Напишем их (после isOnSnake):

function showScore() {
  document.getElementById("score").textContent = score;
}

function placeApple() {
  apple.x = randomCell();
  apple.y = randomCell();
  while (isOnSnake(apple.x, apple.y)) {
    apple.x = randomCell();
    apple.y = randomCell();
  }
}

Цикл while

while значит «пока». while (условие) { … } повторяет код в скобках, пока условие правда. Здесь так: «пока яблоко попало на змейку, перекладывай его в новое случайное место». Как только яблоко окажется на свободной клеточке, условие станет false и цикл закончится.

Цикл for удобен, когда знаешь, сколько раз повторять. А while — когда повторяешь, пока не получится.

Теперь в функции step можно использовать новых помощников:

  if (head.x === apple.x && head.y === apple.y) {
    score = score + 1;
    document.getElementById("score").textContent = score;
    apple.x = randomCell();
    apple.y = randomCell();
    showScore();
    placeApple();
  } else {
    snake.pop();
  }

Шаг 4. Пробел — новая игра

В endGame поднимем флажок и добавим подсказку для игрока:

function endGame() {
  gameOver = true;
  clearInterval(timer);
  ctx.fillStyle = "white";
  ctx.font = "32px sans-serif";
  ctx.textAlign = "center";
  ctx.fillText("Игра окончена", 200, 200);
  ctx.font = "18px sans-serif";
  ctx.fillText("Нажми пробел, чтобы сыграть ещё", 200, 240);
}

А в onKey, после четырёх стрелок, добавь ещё одно условие:

  if (event.key === " " && gameOver) {
    startGame();
  }
}

Клавиша пробела называется " " — просто пробел в кавычках.

⚠️ Зачем проверять gameOver? Попробуй ради эксперимента убрать && gameOver и нажать пробел посреди игры, лучше несколько раз. Змейка поползёт всё быстрее и быстрее! Каждый вызов startGame запускает ещё один таймер, и они начинают работать одновременно. Флажок не даёт начать новую игру, пока старая не закончилась. Верни && gameOver на место.

Сохрани и проверь всё: игра начинается сама, при проигрыше появляются надписи, пробел начинает игру заново, счёт обнуляется.

🎯 Задания
  1. Пусть новая игра начинается и по клавише Enter (её название — "Enter"). Подсказка: ||.
  2. Пусть змейка в начале каждой игры появляется в случайном ряду.
  3. ⭐ Добавь паузу на клавишу P (используй event.code === "KeyP"). Подсказка: заведи переменную paused. При нажатии меняй её на противоположную: paused = !paused; (знак ! значит «не»). А в начале step напиши: если paused, то сразу return.
📄 Весь файл snake.html после этого урока 
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Змейка</title>
</head>
<body>
  <h1>Змейка</h1>
  <p>Счёт: <span id="score">0</span></p>
  <canvas id="game" width="400" height="400"></canvas>
  <script>
    let canvas = document.getElementById("game");
    let ctx = canvas.getContext("2d");
    let size = 20;
    let cells = 20;

    let snake;
    let apple = { x: 0, y: 0 };
    let dx;
    let dy;
    let score;
    let gameOver;
    let timer;

    function clearField() {
      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 400, 400);
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * size, y * size, size - 1, size - 1);
    }

    function randomCell() {
      return Math.floor(Math.random() * cells);
    }

    function isOnSnake(x, y) {
      for (let i = 0; i < snake.length; i++) {
        if (snake[i].x === x && snake[i].y === y) {
          return true;
        }
      }
      return false;
    }

    function showScore() {
      document.getElementById("score").textContent = score;
    }

    function placeApple() {
      apple.x = randomCell();
      apple.y = randomCell();
      while (isOnSnake(apple.x, apple.y)) {
        apple.x = randomCell();
        apple.y = randomCell();
      }
    }

    function draw() {
      clearField();
      drawCell(apple.x, apple.y, "red");
      for (let i = 0; i < snake.length; i++) {
        drawCell(snake[i].x, snake[i].y, "lime");
      }
      drawCell(snake[0].x, snake[0].y, "yellow");
    }

    function endGame() {
      gameOver = true;
      clearInterval(timer);
      ctx.fillStyle = "white";
      ctx.font = "32px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText("Игра окончена", 200, 200);
      ctx.font = "18px sans-serif";
      ctx.fillText("Нажми пробел, чтобы сыграть ещё", 200, 240);
    }

    function step() {
      let head = { x: snake[0].x + dx, y: snake[0].y + dy };

      if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
        endGame();
        return;
      }
      if (isOnSnake(head.x, head.y)) {
        endGame();
        return;
      }

      snake.unshift(head);

      if (head.x === apple.x && head.y === apple.y) {
        score = score + 1;
        showScore();
        placeApple();
      } else {
        snake.pop();
      }

      draw();
    }

    function onKey(event) {
      if (event.key === "ArrowUp" && dy !== 1) {
        dx = 0;
        dy = -1;
      }
      if (event.key === "ArrowDown" && dy !== -1) {
        dx = 0;
        dy = 1;
      }
      if (event.key === "ArrowLeft" && dx !== 1) {
        dx = -1;
        dy = 0;
      }
      if (event.key === "ArrowRight" && dx !== -1) {
        dx = 1;
        dy = 0;
      }
      if (event.key === " " && gameOver) {
        startGame();
      }
    }

    function startGame() {
      snake = [
        { x: 10, y: 10 },
        { x: 9, y: 10 },
        { x: 8, y: 10 }
      ];
      dx = 1;
      dy = 0;
      score = 0;
      gameOver = false;
      showScore();
      placeApple();
      draw();
      timer = setInterval(step, 200);
    }

    document.addEventListener("keydown", onKey);
    startGame();
  </script>
</body>
</html>
