Управление стрелками

Змейка ползёт сама по себе, а пора бы ей слушаться игрока. Научим программу узнавать, какую клавишу нажали, и поворачивать змейку стрелками.

В этом уроке ты:
  • узнаешь, что такое события;
  • «подпишешься» на нажатия клавиш с помощью addEventListener;
  • узнаешь, как называются клавиши-стрелки;
  • научишь змейку поворачивать, но не разворачиваться назад.

События

Когда ты нажимаешь клавишу, двигаешь мышку или щёлкаешь по кнопке, браузер замечает это. Такие случаи называются событиями. Мы можем попросить браузер: «Когда нажмут клавишу, вызови мою функцию».

Добавь в конец скрипта, перед draw();:

function onKey(event) {
  console.log(event.key);
}

document.addEventListener("keydown", onKey);
  • "keydown" — событие «клавишу нажали» (key down — «клавиша вниз»).
  • addEventListener — «добавь слушателя»: теперь при каждом нажатии клавиши браузер будет вызывать нашу функцию onKey.
  • Браузер сам передаёт в функцию параметр event, где записано всё о событии. В event.key лежит название клавиши.

Сохрани, обнови страницу, открой консоль и щёлкни мышкой по странице (чтобы она «слушала» клавиатуру). Понажимай разные клавиши и стрелки. В консоли появятся их названия:

ArrowUp
ArrowDown
ArrowLeft
ArrowRight
a
Enter

Arrow по-английски — «стрелка». Значит, ArrowUp — стрелка вверх.

💡 Клавиши не работают? Если ты нажимаешь клавиши, а в консоли ничего нет, — щёлкни мышкой по самой странице, например по чёрному полю. Клавиши получает то окно или та часть экрана, где был последний щелчок.

Поворачиваем

Теперь вместо console.log будем менять направление dx и dy. Помнишь таблицу направлений из урока 8? Замени функцию onKey:

function onKey(event) {
  console.log(event.key);
  if (event.key === "ArrowUp") {
    dx = 0;
    dy = -1;
  }
  if (event.key === "ArrowDown") {
    dx = 0;
    dy = 1;
  }
  if (event.key === "ArrowLeft") {
    dx = -1;
    dy = 0;
  }
  if (event.key === "ArrowRight") {
    dx = 1;
    dy = 0;
  }
}

Сохрани, обнови, щёлкни по полю и управляй стрелками! 🎮

Название клавиши — это текст, поэтому "ArrowUp" пишется в кавычках. Большие буквы важны: "arrowup" не сработает.

Нельзя назад!

Попробуй: змейка ползёт вправо, а ты нажимаешь стрелку влево. Змейка разворачивается и проползает сама сквозь себя. В настоящей Змейке так нельзя: голова врезалась бы в собственное тело.

Правило такое: повернуть вверх можно, только если змейка сейчас не ползёт вниз. Ползёт вниз — значит, dy равно 1. Добавим это в условие через && («и»):

function onKey(event) {
  if (event.key === "ArrowUp" && dy !== 1) {
    dx = 0;
    dy = -1;
  }
  if (event.key === "ArrowDown" && dy !== -1) {
    dx = 0;
    dy = 1;
  }
  if (event.key === "ArrowLeft" && dx !== 1) {
    dx = -1;
    dy = 0;
  }
  if (event.key === "ArrowRight" && dx !== -1) {
    dx = 1;
    dy = 0;
  }
}

Прочитаем первое условие по-русски: «Если нажата стрелка вверх и змейка не ползёт вниз, то повернуть вверх».

  3. ⭐ Многие игры управляются клавишами W A S D. Добавь их. Но есть хитрость: в русской раскладке event.key для клавиши W будет "ц"! Поэтому используй event.code — это «место» клавиши на клавиатуре, оно не зависит от раскладки: "KeyW", "KeyA", "KeyS", "KeyD". Подсказка: условие можно расширить через ||: 
    if ((event.key === "ArrowUp" || event.code === "KeyW") && dy !== 1) {
📄 Весь файл snake.html после этого урока 
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Змейка</title>
</head>
<body>
  <h1>Змейка</h1>
  <canvas id="game" width="400" height="400"></canvas>
  <script>
    let canvas = document.getElementById("game");
    let ctx = canvas.getContext("2d");
    let size = 20;
    let cells = 20;

    let snake = [
      { x: 10, y: 10 },
      { x: 9, y: 10 },
      { x: 8, y: 10 }
    ];
    let apple = { x: 15, y: 10 };
    let dx = 1;
    let dy = 0;

    function clearField() {
      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 400, 400);
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * size, y * size, size - 1, size - 1);
    }

    function draw() {
      clearField();
      drawCell(apple.x, apple.y, "red");
      for (let i = 0; i < snake.length; i++) {
        drawCell(snake[i].x, snake[i].y, "lime");
      }
      drawCell(snake[0].x, snake[0].y, "yellow");
    }

    function step() {
      let head = { x: snake[0].x + dx, y: snake[0].y + dy };

      if (head.x >= cells) {
        head.x = 0;
      }
      if (head.x < 0) {
        head.x = cells - 1;
      }
      if (head.y >= cells) {
        head.y = 0;
      }
      if (head.y < 0) {
        head.y = cells - 1;
      }

      snake.unshift(head);
      snake.pop();
      draw();
    }

    function onKey(event) {
      if (event.key === "ArrowUp" && dy !== 1) {
        dx = 0;
        dy = -1;
      }
      if (event.key === "ArrowDown" && dy !== -1) {
        dx = 0;
        dy = 1;
      }
      if (event.key === "ArrowLeft" && dx !== 1) {
        dx = -1;
        dy = 0;
      }
      if (event.key === "ArrowRight" && dx !== -1) {
        dx = 1;
        dy = 0;
      }
    }

    document.addEventListener("keydown", onKey);
    draw();
    setInterval(step, 200);
  </script>
</body>
</html>
