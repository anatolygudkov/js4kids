Nueva partida

Pierdes y tienes que pulsar F5… ¡Qué incómodo! Vamos a hacerlo como en los juegos de verdad: pulsas espacio y empieza una partida nueva. Para eso vamos a reorganizar un poco el programa. Además, impediremos que la manzana aparezca dentro de la serpiente.

En esta lección vas a:
  • reunir todo lo necesario para empezar una partida en una función startGame;
  • añadir una variable-bandera llamada gameOver;
  • hacer que la barra espaciadora reinicie el juego;
  • conocer el bucle while.

Qué significa «empezar de nuevo»

Cuando empieza una partida nueva hay que:

  1. volver a poner la serpiente de tres trozos en el centro del campo;
  2. dirigirla hacia la derecha;
  3. poner los puntos a cero;
  4. poner la manzana en un sitio nuevo;
  5. dibujar el campo y poner en marcha el temporizador.

Ahora casi todo esto ocurre al principio del programa, cuando creamos las variables. Lo vamos a pasar a una función startGame («empezar partida») para poder llamarla todas las veces que queramos.

💡 Esto se llama refactorizar Cuando quien programa reorganiza el código para que sea más cómodo, pero el juego sigue funcionando igual, se llama refactorizar. Es una parte normal del trabajo. Ve sin prisa y, después de cada paso, comprueba que el juego arranca.

Paso 1. Cajas vacías

Las variables se quedan arriba, pero los valores se los daremos dentro de startGame. Si escribes let sin =, sale una caja vacía. Añadimos también dos nuevas: gameOver y timer.

let canvas = document.getElementById("game");
let ctx = canvas.getContext("2d");
let size = 20;
let cells = 20;

let snake = [
  { x: 10, y: 10 },
  { x: 9, y: 10 },
  { x: 8, y: 10 }
];
let apple = { x: 15, y: 10 };
let dx = 1;
let dy = 0;
let score = 0;
let snake;
let apple = { x: 0, y: 0 };
let dx;
let dy;
let score;
let gameOver;
let timer;

gameOver («fin del juego» en inglés) es una bandera: guarda true o false. Con ella el programa sabe si la partida sigue o ya ha terminado.

Paso 2. La función startGame

Añádela al final del todo, después de la función onKey:

function startGame() {
  snake = [
    { x: 10, y: 10 },
    { x: 9, y: 10 },
    { x: 8, y: 10 }
  ];
  dx = 1;
  dy = 0;
  score = 0;
  gameOver = false;
  showScore();
  placeApple();
  draw();
  timer = setInterval(step, 200);
}

Y las últimas líneas del programa quedan así:

document.addEventListener("keydown", onKey);
draw();
let timer = setInterval(step, 200);
startGame();

Fíjate: dentro de startGame escribimos snake = … y timer = … sin let, porque las cajas ya están creadas arriba. Si pones let, sale una caja nueva que solo vive dentro de la función, y el resto del programa no la ve.

Paso 3. Ayudantes nuevos

En startGame hemos llamado a dos funciones que todavía no existen: showScore y placeApple. Vamos a escribirlas (después de isOnSnake):

function showScore() {
  document.getElementById("score").textContent = score;
}

function placeApple() {
  apple.x = randomCell();
  apple.y = randomCell();
  while (isOnSnake(apple.x, apple.y)) {
    apple.x = randomCell();
    apple.y = randomCell();
  }
}

El bucle while

while significa «mientras». while (condición) { … } repite el código de las llaves mientras la condición sea verdad. Aquí quiere decir: «mientras la manzana esté encima de la serpiente, llévala a otro sitio al azar». En cuanto la manzana cae en una casilla libre, la condición se vuelve false y el bucle termina.

El bucle for va bien cuando sabes cuántas veces repetir. Y while, cuando repites hasta que salga bien.

Ahora la función step puede usar los ayudantes nuevos:

  if (head.x === apple.x && head.y === apple.y) {
    score = score + 1;
    document.getElementById("score").textContent = score;
    apple.x = randomCell();
    apple.y = randomCell();
    showScore();
    placeApple();
  } else {
    snake.pop();
  }

Paso 4. Espacio: nueva partida

En endGame levantamos la bandera y añadimos una pista para quien juega:

function endGame() {
  gameOver = true;
  clearInterval(timer);
  ctx.fillStyle = "white";
  ctx.font = "32px sans-serif";
  ctx.textAlign = "center";
  ctx.fillText("Fin del juego", 200, 200);
  ctx.font = "18px sans-serif";
  ctx.fillText("Pulsa espacio para jugar otra vez", 200, 240);
}

Y en onKey, después de las cuatro flechas, añade una condición más:

  if (event.key === " " && gameOver) {
    startGame();
  }
}

La barra espaciadora se llama " ": un espacio entre comillas.

⚠️ ¿Para qué comprobar gameOver? Como experimento, quita && gameOver y pulsa espacio en mitad de una partida, mejor varias veces. ¡La serpiente irá cada vez más rápido! Cada llamada a startGame pone en marcha otro temporizador, y todos funcionan a la vez. La bandera impide empezar una partida nueva hasta que la anterior haya terminado. Vuelve a poner && gameOver.

Guarda y compruébalo todo: el juego empieza solo, al perder salen los mensajes, el espacio vuelve a empezar la partida y los puntos se ponen a cero.

📄 El archivo snake.html completo después de esta lección 
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Serpiente</title>
</head>
<body>
  <h1>Serpiente</h1>
  <p>Puntos: <span id="score">0</span></p>
  <canvas id="game" width="400" height="400"></canvas>
  <script>
    let canvas = document.getElementById("game");
    let ctx = canvas.getContext("2d");
    let size = 20;
    let cells = 20;

    let snake;
    let apple = { x: 0, y: 0 };
    let dx;
    let dy;
    let score;
    let gameOver;
    let timer;

    function clearField() {
      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 400, 400);
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * size, y * size, size - 1, size - 1);
    }

    function randomCell() {
      return Math.floor(Math.random() * cells);
    }

    function isOnSnake(x, y) {
      for (let i = 0; i < snake.length; i++) {
        if (snake[i].x === x && snake[i].y === y) {
          return true;
        }
      }
      return false;
    }

    function showScore() {
      document.getElementById("score").textContent = score;
    }

    function placeApple() {
      apple.x = randomCell();
      apple.y = randomCell();
      while (isOnSnake(apple.x, apple.y)) {
        apple.x = randomCell();
        apple.y = randomCell();
      }
    }

    function draw() {
      clearField();
      drawCell(apple.x, apple.y, "red");
      for (let i = 0; i < snake.length; i++) {
        drawCell(snake[i].x, snake[i].y, "lime");
      }
      drawCell(snake[0].x, snake[0].y, "yellow");
    }

    function endGame() {
      gameOver = true;
      clearInterval(timer);
      ctx.fillStyle = "white";
      ctx.font = "32px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText("Fin del juego", 200, 200);
      ctx.font = "18px sans-serif";
      ctx.fillText("Pulsa espacio para jugar otra vez", 200, 240);
    }

    function step() {
      let head = { x: snake[0].x + dx, y: snake[0].y + dy };

      if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
        endGame();
        return;
      }
      if (isOnSnake(head.x, head.y)) {
        endGame();
        return;
      }

      snake.unshift(head);

      if (head.x === apple.x && head.y === apple.y) {
        score = score + 1;
        showScore();
        placeApple();
      } else {
        snake.pop();
      }

      draw();
    }

    function onKey(event) {
      if (event.key === "ArrowUp" && dy !== 1) {
        dx = 0;
        dy = -1;
      }
      if (event.key === "ArrowDown" && dy !== -1) {
        dx = 0;
        dy = 1;
      }
      if (event.key === "ArrowLeft" && dx !== 1) {
        dx = -1;
        dy = 0;
      }
      if (event.key === "ArrowRight" && dx !== -1) {
        dx = 1;
        dy = 0;
      }
      if (event.key === " " && gameOver) {
        startGame();
      }
    }

    function startGame() {
      snake = [
        { x: 10, y: 10 },
        { x: 9, y: 10 },
        { x: 8, y: 10 }
      ];
      dx = 1;
      dy = 0;
      score = 0;
      gameOver = false;
      showScore();
      placeApple();
      draw();
      timer = setInterval(step, 200);
    }

    document.addEventListener("keydown", onKey);
    startGame();
  </script>
</body>
</html>
