D4 Write code ⭐⭐⭐

E4 Find the bug ⭐⭐ 

    
This program freezes. Don't run it! Find the mistake by reading.

    
for (let i = 0; i < 5; i--) {
  console.log(i);
}

    
Solve this one in your own file in Geany — it can't run here.

E5 Write code ⭐⭐⭐ 

    
Draw an 8 × 8 chessboard of 50 × 50 squares — it fills the whole canvas.

    
 

    
💡 Hint
Two loops, one inside the other: over y and over x. The % sign gives the remainder of a division: (x + y) % 2 is 0 or 1 — that's the alternating colour.

    
✅ Show answer
The outer loop goes through the rows, the inner one through the squares in a row. Neighbouring squares differ by 1 in x + y, so the colours alternate. (If you haven't done if yet, come back after lesson 9.)
let canvas = document.getElementById("game");
let ctx = canvas.getContext("2d");

for (let y = 0; y < 8; y++) {
  for (let x = 0; x < 8; x++) {
    if ((x + y) % 2 === 0) {
      ctx.fillStyle = "white";
    } else {
      ctx.fillStyle = "black";
    }
    ctx.fillRect(x * 50, y * 50, 50, 50);
  }
}

H4 Write code ⭐⭐ 

    
Finish the wrap(n) function for the portals: if n has left the 0…19 field, bring it back on the other side. It should print 0, 19 and 7.

    
 

    
✅ Show answer
The same two ifs as the portals in lesson 9, just packed into a function. Now you can call it for both x and y.
let cells = 20;

function wrap(n) {
  if (n >= cells) {
    n = 0;
  }
  if (n < 0) {
    n = cells - 1;
  }
  return n;
}

console.log(wrap(20));
console.log(wrap(-1));
console.log(wrap(7));

  

  

    
H5 Write code ⭐⭐⭐ 

    
Make the arrow keys move the square around the canvas by 20 pixels. Run it, click on the picture, and press the arrows.

    
 

    
✅ Show answer
Each arrow changes x or y, and then everything is redrawn. ⭐ Can you stop the square from leaving the field?
let canvas = document.getElementById("game");
let ctx = canvas.getContext("2d");

let x = 180;
let y = 180;

function draw() {
  ctx.fillStyle = "black";
  ctx.fillRect(0, 0, 400, 400);
  ctx.fillStyle = "lime";
  ctx.fillRect(x, y, 40, 40);
}

function onKey(event) {
  if (event.key === "ArrowUp") {
    y = y - 20;
  }
  if (event.key === "ArrowDown") {
    y = y + 20;
  }
  if (event.key === "ArrowLeft") {
    x = x - 20;
  }
  if (event.key === "ArrowRight") {
    x = x + 20;
  }
  draw();
}

document.addEventListener("keydown", onKey);
draw();

  

  

    
H6 Find the bug ⭐⭐⭐ 

    
Run the program and read the error. Why does it happen, when nobody has pressed a key yet?

    
 

    
✅ Show answer
Because of the brackets, onKey() is called straight away, without an event, so event is empty (undefined). The listener needs the function itself: onKey without brackets. It's the same mistake as with setInterval.
function onKey(event) {
  console.log(event.key);
}

document.addEventListener("keydown", onKey);

K3 Write code ⭐⭐ 

    
Write a function contains(list, value) that answers true if value is in the array and false if not. It's the same idea as isOnSnake.

    
 

    
✅ Show answer
return true goes inside the loop, as soon as it's found. return false goes after the loop, once everything has been checked.
let numbers = [3, 8, 5];

function contains(list, value) {
  for (let i = 0; i < list.length; i++) {
    if (list[i] === value) {
      return true;
    }
  }
  return false;
}

console.log(contains(numbers, 8));
console.log(contains(numbers, 4));
