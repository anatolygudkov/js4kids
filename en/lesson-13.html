New game

Lose, then press F5… Not very handy! Let's do it like in real games: press Space and a new game starts. For that we'll rearrange the program a little. We'll also stop the apple from appearing inside the snake.

In this lesson you will:
  • collect everything needed to start a game into a startGame function;
  • add a flag variable called gameOver;
  • make Space restart the game;
  • meet the while loop.

What "start again" means

When a new game begins, we need to:

  1. put the three-piece snake back in the middle of the field;
  2. point it to the right;
  3. reset the score to zero;
  4. put the apple in a new place;
  5. draw the field and start the timer.

Right now almost all of this happens at the very start of the program, when we create the variables. Let's move it into a startGame function so it can be called as many times as we like.

💡 This is called refactoring When programmers rearrange code to make it more convenient, while the game still works the same way, it's called refactoring. It's a normal part of the job. Take your time, and after each step check that the game still starts.

Step 1. Empty boxes

We'll keep the variables at the top, but put values into them inside startGame. If you write let without =, you get an empty box. Let's also add two new ones: gameOver and timer.

let canvas = document.getElementById("game");
let ctx = canvas.getContext("2d");
let size = 20;
let cells = 20;

let snake = [
  { x: 10, y: 10 },
  { x: 9, y: 10 },
  { x: 8, y: 10 }
];
let apple = { x: 15, y: 10 };
let dx = 1;
let dy = 0;
let score = 0;
let snake;
let apple = { x: 0, y: 0 };
let dx;
let dy;
let score;
let gameOver;
let timer;

gameOver is a flag: it holds true or false. It tells the program whether the game is still going or has already finished.

Step 2. The startGame function

Add it at the very end, after the onKey function:

function startGame() {
  snake = [
    { x: 10, y: 10 },
    { x: 9, y: 10 },
    { x: 8, y: 10 }
  ];
  dx = 1;
  dy = 0;
  score = 0;
  gameOver = false;
  showScore();
  placeApple();
  draw();
  timer = setInterval(step, 200);
}

And the last lines of the program become:

document.addEventListener("keydown", onKey);
draw();
let timer = setInterval(step, 200);
startGame();

Notice: inside startGame we write snake = … and timer = … without let, because the boxes were already created at the top. If you add let, you get a new box that only lives inside the function, and the rest of the program can't see it.

Step 3. New helpers

In startGame we called two functions that don't exist yet: showScore and placeApple. Let's write them (after isOnSnake):

function showScore() {
  document.getElementById("score").textContent = score;
}

function placeApple() {
  apple.x = randomCell();
  apple.y = randomCell();
  while (isOnSnake(apple.x, apple.y)) {
    apple.x = randomCell();
    apple.y = randomCell();
  }
}

The while loop

while repeats the code in its curly brackets as long as the condition is true. Here it means: "while the apple is on the snake, move it to a new random place". As soon as the apple lands on a free cell, the condition becomes false and the loop ends.

A for loop is handy when you know how many times to repeat. while is for repeating until it works out.

Now the step function can use the new helpers:

  if (head.x === apple.x && head.y === apple.y) {
    score = score + 1;
    document.getElementById("score").textContent = score;
    apple.x = randomCell();
    apple.y = randomCell();
    showScore();
    placeApple();
  } else {
    snake.pop();
  }

Step 4. Space starts a new game

In endGame we'll raise the flag and add a hint for the player:

function endGame() {
  gameOver = true;
  clearInterval(timer);
  ctx.fillStyle = "white";
  ctx.font = "32px sans-serif";
  ctx.textAlign = "center";
  ctx.fillText("Game over", 200, 200);
  ctx.font = "18px sans-serif";
  ctx.fillText("Press Space to play again", 200, 240);
}

And in onKey, after the four arrows, add one more condition:

  if (event.key === " " && gameOver) {
    startGame();
  }
}

The Space key is called " " — just a space in quotes.

⚠️ Why check gameOver? As an experiment, try removing && gameOver and pressing Space in the middle of a game, a few times if you can. The snake will go faster and faster! Each call to startGame starts another timer, and they all run at once. The flag stops a new game from starting until the old one has finished. Put && gameOver back.

Save and check everything: the game starts by itself, the messages appear when you lose, Space starts the game again, and the score resets.

📄 The whole snake.html file after this lesson 
<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>Snake</title>
</head>
<body>
  <h1>Snake</h1>
  <p>Score: <span id="score">0</span></p>
  <canvas id="game" width="400" height="400"></canvas>
  <script>
    let canvas = document.getElementById("game");
    let ctx = canvas.getContext("2d");
    let size = 20;
    let cells = 20;

    let snake;
    let apple = { x: 0, y: 0 };
    let dx;
    let dy;
    let score;
    let gameOver;
    let timer;

    function clearField() {
      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 400, 400);
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * size, y * size, size - 1, size - 1);
    }

    function randomCell() {
      return Math.floor(Math.random() * cells);
    }

    function isOnSnake(x, y) {
      for (let i = 0; i < snake.length; i++) {
        if (snake[i].x === x && snake[i].y === y) {
          return true;
        }
      }
      return false;
    }

    function showScore() {
      document.getElementById("score").textContent = score;
    }

    function placeApple() {
      apple.x = randomCell();
      apple.y = randomCell();
      while (isOnSnake(apple.x, apple.y)) {
        apple.x = randomCell();
        apple.y = randomCell();
      }
    }

    function draw() {
      clearField();
      drawCell(apple.x, apple.y, "red");
      for (let i = 0; i < snake.length; i++) {
        drawCell(snake[i].x, snake[i].y, "lime");
      }
      drawCell(snake[0].x, snake[0].y, "yellow");
    }

    function endGame() {
      gameOver = true;
      clearInterval(timer);
      ctx.fillStyle = "white";
      ctx.font = "32px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText("Game over", 200, 200);
      ctx.font = "18px sans-serif";
      ctx.fillText("Press Space to play again", 200, 240);
    }

    function step() {
      let head = { x: snake[0].x + dx, y: snake[0].y + dy };

      if (head.x < 0 || head.x >= cells || head.y < 0 || head.y >= cells) {
        endGame();
        return;
      }
      if (isOnSnake(head.x, head.y)) {
        endGame();
        return;
      }

      snake.unshift(head);

      if (head.x === apple.x && head.y === apple.y) {
        score = score + 1;
        showScore();
        placeApple();
      } else {
        snake.pop();
      }

      draw();
    }

    function onKey(event) {
      if (event.key === "ArrowUp" && dy !== 1) {
        dx = 0;
        dy = -1;
      }
      if (event.key === "ArrowDown" && dy !== -1) {
        dx = 0;
        dy = 1;
      }
      if (event.key === "ArrowLeft" && dx !== 1) {
        dx = -1;
        dy = 0;
      }
      if (event.key === "ArrowRight" && dx !== -1) {
        dx = 1;
        dy = 0;
      }
      if (event.key === " " && gameOver) {
        startGame();
      }
    }

    function startGame() {
      snake = [
        { x: 10, y: 10 },
        { x: 9, y: 10 },
        { x: 8, y: 10 }
      ];
      dx = 1;
      dy = 0;
      score = 0;
      gameOver = false;
      showScore();
      placeApple();
      draw();
      timer = setInterval(step, 200);
    }

    document.addEventListener("keydown", onKey);
    startGame();
  </script>
</body>
</html>
